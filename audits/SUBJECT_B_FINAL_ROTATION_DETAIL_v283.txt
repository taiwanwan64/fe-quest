function buildBFinal(){
  // 40問の本番型アルゴリズムプールから、
  // 10分野を最低1問ずつ・標準8問/応用8問になるよう16問を構成する。
  const selected=[];
  const used=new Set();

  B_FINAL_ALGO_DOMAINS.forEach(domain=>{
    const desired=B_FINAL_APPLIED_DOMAINS.has(domain)?'応用':'標準';
    let pool=B_EXAM_ALGO_ITEMS.filter(x=>x.domain===domain&&x.level===desired&&!used.has(x.id));
    if(!pool.length)pool=B_EXAM_ALGO_ITEMS.filter(x=>x.domain===domain&&!used.has(x.id));
    const pick=pickFinalAlgoCandidate(pool,selected);
    if(pick){selected.push(pick);used.add(pick.id);}
  });

  const fillLevel=(level,target)=>{
    while(selected.filter(x=>x.level===level).length<target){
      const pool=B_EXAM_ALGO_ITEMS.filter(x=>x.level===level&&!used.has(x.id));
      const pick=pickFinalAlgoCandidate(pool,selected);
      if(!pick)break;
      selected.push(pick);used.add(pick.id);
    }
  };
  fillLevel('応用',8);
  fillLevel('標準',8);

  // 念のため不足時だけ未使用問題で補完する。
  while(selected.length<16){
    const pick=pickFinalAlgoCandidate(B_EXAM_ALGO_ITEMS.filter(x=>!used.has(x.id)),selected);
    if(!pick)break;
    selected.push(pick);used.add(pick.id);
  }
  const algo=selected.slice(0,16).map(makeFinalAlgoExam);

  // 情報セキュリティ4問は、ログ読解2問＋ケース判断2問を固定。
  const secUsed=new Set();
  const secLogs=pickSecurityForFinal(SECURITY_SCENARIOS.filter(s=>s.log),2,secUsed);
  const secCases=pickSecurityForFinal(SECURITY_SCENARIOS.filter(s=>!s.log),2,secUsed);
  const secPool=[...secLogs,...secCases].map(makeFinalSecurity);

  return shuffled([...algo,...secPool]);
}

function bFinalRepairTraceFloorV208(items){
  const out=[...items];
  let algo=out.filter(x=>x?.kind==='algo');
  let highCount=bFinalHighTraceCountV208(algo);

  while(highCount<B_FINAL_HIGH_TRACE_FLOOR_V208){
    const selectedIds=new Set(algo.map(x=>x.sourceId));
    const rawById=new Map(B_EXAM_ALGO_ITEMS.map(x=>[x.id,x]));
    const rawSelected=algo.map(x=>rawById.get(x.sourceId)).filter(Boolean);
    const formatCounts={};
    rawSelected.forEach(x=>{const f=x.format||'トレース';formatCounts[f]=(formatCounts[f]||0)+1;});
    const swaps=[];

    algo.forEach(victim=>{
      if(B_FINAL_HIGH_TRACE_IDS_V208.has(victim.sourceId))return;
      const rawVictim=rawById.get(victim.sourceId);
      if(!rawVictim)return;
      B_EXAM_ALGO_ITEMS
        .filter(candidate=>B_FINAL_HIGH_TRACE_IDS_V208.has(candidate.id)&&!selectedIds.has(candidate.id)&&candidate.domain===rawVictim.domain&&candidate.level===rawVictim.level)
        .forEach(candidate=>{
          const candidateFormat=candidate.format||'トレース';
          const victimFormat=rawVictim.format||'トレース';
          const prospectiveFormatLoad=(formatCounts[candidateFormat]||0)-(candidateFormat===victimFormat?1:0);
          swaps.push({
            victim,
            rawVictim,
            candidate,
            candidateSeen:bFinalAlgoSeen(candidate),
            prospectiveFormatLoad,
            victimSeen:bFinalAlgoSeen(rawVictim)
          });
        });
    });

    swaps.sort((a,b)=>
      a.candidateSeen-b.candidateSeen ||
      a.prospectiveFormatLoad-b.prospectiveFormatLoad ||
      b.victimSeen-a.victimSeen ||
      Math.random()-.5
    );
    const swap=swaps[0];
    if(!swap)break;
    const pos=out.findIndex(x=>x?.kind==='algo'&&x.sourceId===swap.victim.sourceId);
    if(pos<0)break;
    out[pos]=makeFinalAlgoExam(swap.candidate);
    algo=out.filter(x=>x?.kind==='algo');
    highCount=bFinalHighTraceCountV208(algo);
  }
  return out;
}
