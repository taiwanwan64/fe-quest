    const pick=pickFinalAlgoCandidate(pool,selected);
    if(pick){selected.push(pick);used.add(pick.id);}
  });

  const fillLevel=(level,target)=>{
    while(selected.filter(x=>x.level===level).length<target){
      const pool=B_EXAM_ALGO_ITEMS.filter(x=>x.level===level&&!used.has(x.id));
      const pick=pickFinalAlgoCandidate(pool,selected);
