function(attempt,earned){
  const out=__renderBFinalResultBeforeV245(attempt,earned);
  const wrong=(attempt?.details||[]).filter(d=>!d.ok);
  const rows=[...document.querySelectorAll('.bfinal-review-item.wrong')];
  rows.forEach((row,i)=>{
    const detail=wrong[i];
    if(detail?.kind!=='security')return;
    const key=bFinalMistakeKey(detail);
    const stored=profile.bFinalMistakeStats?.[key]?.lastReason||'';
    const canonical=subjectBSecurityReasonCanonicalV245(stored);
    row.querySelectorAll('[data-bfreason]').forEach(btn=>{
      if(btn.dataset.bfreason==='トレースミス'){
        btn.dataset.bfreason='手順の追い違い';
        btn.textContent='手順の追い違い';
      }else if(btn.dataset.bfreason==='コード理解'){
        btn.dataset.bfreason='対策の理解';
        btn.textContent='対策の理解';
      }
      btn.classList.toggle('picked',btn.dataset.bfreason===canonical);
    });
  });
  return out;
}

function renderPracticeNextCard(){
  const title=document.getElementById('bPracticeNextTitle'),desc=document.getElementById('bPracticeNextDesc'),action=document.getElementById('bPracticeNextAction');
  if(!title||!desc||!action)return;
  const rec=subjectBHubRecommendation();
  if(rec.stage<3){
    title.textContent=`先に ${rec.title}`;desc.textContent='実戦演習へ進む前に、未完了の基礎演習を1つ進めるのがおすすめです。';action.textContent='基礎演習へ →';
  }else if(rec.mode==='final'){
    title.textContent='短い実戦演習は一通り完了しました';desc.textContent='100分を確保できる日は、科目B 総合実戦で20問を通して確認できます。';action.textContent='総合実戦を確認 →';
  }else if(rec.mode==='securityMock'){
    title.textContent='セキュリティ ミニ模試';desc.textContent='次はケース判断を8問・20分でヒントなし確認します。';action.textContent='セキュリティ実戦へ →';
  }else if(rec.mode==='miniMock'){
    title.textContent='アルゴリズム ミニ模試';desc.textContent='8問・40分で、途中状態を連続して読み切れるか確認します。';action.textContent='ミニ模試を始める →';
  }else{
    title.textContent=rec.title||'複合問題を1セット';desc.textContent=rec.desc||'長いコードを3問で読みます。';action.textContent='複合問題を始める →';
  }
  action.onclick=()=>launchSubjectBRecommendation(rec);
}
