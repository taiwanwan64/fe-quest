    assert(CORE_A_CHALLENGE_QUESTIONS.filter(q=>q.angle==='comparison').every(q=>Array.isArray(q.comparisonTopicIds)&&q.comparisonTopicIds.length===2),'Comparison challenges need an explicit two-topic scope');
    const requiredDom=['home','map','weak','problems','plan','coverage','mock','lesson','trace','settingsBtn'];
    requiredDom.forEach(id=>assert(!!document.getElementById(id),`Required DOM missing: ${id}`));

    if(typeof validateGeneratedVariants==='function'){
      const n=validateGeneratedVariants();
      assert(n>0,'Variant generator validation returned 0');
    }
  }catch(e){
    errors.push('Self-check exception: '+(e?.message||String(e)));
  }

  return {ok:errors.length===0,errors,warnings,checkedAt:new Date().toISOString()};
}
