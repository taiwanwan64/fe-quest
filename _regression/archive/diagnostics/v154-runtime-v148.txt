(() => {
  const NODE_SMOKE = () => globalThis.__FEQUEST_RUNTIME_SMOKE__===true || /node-runtime-smoke/i.test(String(globalThis.navigator?.userAgent||''));
  const currentCheck = (id,message,test,legacyAssertions=0) => ({id,message,test,legacyAssertions});

  const MIGRATION_GROUPS = [
    'judgment-distractor-cue','security-distractor-cue','structural-answer-cue','strong-answer-length-cue',
    'focused-exercise-flow','core-course-map-behavior','core-polite-prose','plan-interaction-helpers'
  ];

  const V148_EXTRA_UI_CONTRACTS = [
    {id:'v111-plan-priority',message:'v111 plan next-action / compact details missing',test:()=>typeof renderPlanFocus==='function'&&!!document.getElementById('planFocusCard')&&!!document.getElementById('planDetailsToggle')},
    {id:'v111-analytics-priority',message:'v111 analytics priority / progressive disclosure missing',test:()=>!!document.querySelector('#history .analytics-priority-card')&&!!document.getElementById('analyticsDetailsToggle')},
    {id:'v111-data-recovery-folds',message:'v111 data/recovery progressive disclosure missing',test:()=>!!document.querySelector('#pwaHealthCard .data-maintenance-fold')&&!!document.querySelector('#pwaHealthCard .recovery-fold')},
    {id:'v113-back-accessibility',message:'v113 icon-only back buttons need accessible names',test:()=>[...document.querySelectorAll('button.back')].every(b=>b.hasAttribute('aria-label'))},
    {id:'v114-weak-primary-action',message:'v114 weak-analysis priority must have a direct action',test:()=>!!document.getElementById('weakTopAction')},
    {id:'v112-desktop-daily-action',message:'v112 desktop daily side action missing',test:()=>!!document.getElementById('rightDailyAction')&&!!document.getElementById('rightDailyProgress')},
    {id:'v115-ai-header-location',message:'v115 BIT先生 must live in the top-right header',test:()=>!!document.querySelector('header #aiFab.ai-header-btn')},
    {id:'v119-focused-exercise-cta',message:'v119 focused exercise CTA contract missing',test:()=>!!document.getElementById('problems')&&!!document.querySelector('.quiz-actions')&&!!document.getElementById('quizSubmit')},
    {id:'v110-subject-b-next-cards',message:'v110 Subject B next-action cards missing',test:()=>!!document.getElementById('subjectBNextCard')&&!!document.getElementById('subjectBProgressStrip')&&!!document.getElementById('bTraceNextCard')&&!!document.getElementById('secNextCard')&&!!document.getElementById('bPracticeNextCard')},
    {id:'v102-legacy-quest-copy',message:'v102 legacy quest wording remains visible',test:()=>!document.body.innerText.includes('今日のクエスト')&&!document.body.innerText.includes('クエスト完了')}
  ];

  function v148CueAudit(){
    const giveawayActionCue=/(だけを(?:確認|変更|実施|行|監視|見る|使|管理|再実行)|だけで(?:判断|決定|対応|運用)|だけ(?:確認|変更|実施|監視|を見る|を使う)|必ず|常に|一切|無条件|絶対)/;
    const incompleteActionCue=/(?:確認|調査|対応|変更|保全|隔離|復元|管理|実施|適用|監視).{0,14}(?:は行わない|を行わない|しない|後回し|省略)|(?:だけ|のみ).{0,14}(?:確認|調査|対応|変更|判断|実施|監視)/;
    const judgmentOptions=QUESTION_BANK.filter(q=>q.cognitiveLevel==='判断').flatMap(q=>(q.options||[]).map((o,i)=>({id:q.id,i,o,a:q.a})));
    const securityOptions=SECURITY_SCENARIOS.flatMap(s=>s.steps.flatMap((q,qi)=>q.options.map((o,i)=>({id:s.id+':'+qi,i,o,a:q.a}))));
    const judgmentGiveaways=judgmentOptions.filter(x=>x.i!==x.a&&giveawayActionCue.test(String(x.o)));
    const securityGiveaways=securityOptions.filter(x=>x.i!==x.a&&giveawayActionCue.test(String(x.o)));
    const judgmentIncomplete=judgmentOptions.filter(x=>x.i!==x.a&&incompleteActionCue.test(String(x.o)));
    const securityIncomplete=securityOptions.filter(x=>x.i!==x.a&&incompleteActionCue.test(String(x.o)));
    const optionComponentCount=o=>(String(o).match(/[、・]|など|および|ならびに/g)||[]).length;
    const judgmentAndSecurity=[...QUESTION_BANK.filter(q=>q.cognitiveLevel==='判断').map(q=>({id:q.id,options:q.options,a:q.a})),...SECURITY_SCENARIOS.flatMap(s=>s.steps.map((q,qi)=>({id:s.id+':'+qi,options:q.options,a:q.a})))];
    const structuralCueFlags=judgmentAndSecurity.filter(x=>{const cs=x.options.map(optionComponentCount);return cs[x.a]>=3&&Math.max(...cs.filter((_,i)=>i!==x.a))<=1;});
    const correctOnlyNado=judgmentAndSecurity.filter(x=>String(x.options[x.a]).includes('など')&&x.options.every((o,i)=>i===x.a||!String(o).includes('など')));
    const strongLengthCue=QUESTION_BANK.filter(q=>{
      const lens=q.options.map(x=>String(x).replace(/\s+/g,'').length);
      const correct=lens[q.a];const other=lens.filter((_x,i)=>i!==q.a);const avg=other.reduce((a,b)=>a+b,0)/Math.max(1,other.length);
      return (correct>=avg*1.5&&correct-avg>=15)||(avg>=correct*1.65&&avg-correct>=18);
    });
    return {judgmentGiveaways,securityGiveaways,judgmentIncomplete,securityIncomplete,structuralCueFlags,correctOnlyNado,strongLengthCue};
  }

  function v148AdditionalCurrentChecks(){
    return [
      currentCheck('judgment-distractor-cue','Judgment distractors must not expose wrong answers through absolute or deliberately incomplete action wording',()=>{const a=v148CueAudit();return a.judgmentGiveaways.length===0&&a.judgmentIncomplete.length===0;},2),
      currentCheck('security-distractor-cue','Security distractors must not expose wrong answers through absolute or deliberately incomplete action wording',()=>{const a=v148CueAudit();return a.securityGiveaways.length===0&&a.securityIncomplete.length===0;},2),
      currentCheck('structural-answer-cue','Judgment/security correct answers must not be uniquely signaled by comprehensive-list structure or 「など」',()=>{const a=v148CueAudit();return a.structuralCueFlags.length===0&&a.correctOnlyNado.length===0;},2),
      currentCheck('strong-answer-length-cue','No Subject A answer should retain a strong correct-vs-distractor length cue',()=>v148CueAudit().strongLengthCue.length===0,1),
      currentCheck('focused-exercise-flow','Focused exercise hub logic must remain available as the primary practice entry',()=>typeof openProblemsHub==='function'&&typeof startQuiz==='function'&&String(openProblemsHub).includes("getElementById('problems')")&&String(openProblemsHub).includes('renderExerciseEntry'),1),
      currentCheck('core-course-map-behavior','Core A course map must not auto-expand the first incomplete chapter',()=>typeof renderCoreCourseMap==='function'&&!String(renderCoreCourseMap).includes("ch===focusChapter)?'open'"),1),
      currentCheck('core-polite-prose','Core A explanatory prose must retain polite-form and caution normalization',()=>typeof politeCoreProse==='function'&&typeof politeCoreCautionProse==='function'&&politeCoreProse('CPU性能はクロック周波数だけでは決まらない。').endsWith('決まりません。')&&politeCoreProse('違いを見る。')==='違いを見ます。'&&politeCoreCautionProse('用語を混同しない。')==='用語を混同しないようにします。'&&!politeCoreProse('処理は高速だが、条件を確認する。').includes('だが、'),4),
      currentCheck('plan-interaction-helpers','Plan priority rendering and compact-detail toggling must remain functional',()=>typeof renderPlanFocus==='function'&&typeof setPlanDetailsOpen==='function'&&String(renderPlanFocus).includes("getElementById('planFocusTitle')")&&String(setPlanDetailsOpen).includes("getElementById('planDetailsToggle')"),2)
    ];
  }

  function v148EvaluateCurrentContract(){
    const base=v147EvaluateCurrentContract();
    const baseResults=(base.results||[]).map(r=>r.id==='version-schema'?{...r,message:'Current version/schema contract',pass:PROFILE_SCHEMA_VERSION===5&&APP_VERSION==='v148',error:undefined}:{...r});
    const extras=v148AdditionalCurrentChecks().map(c=>{try{return {id:c.id,message:c.message,pass:!!c.test(),legacyAssertions:c.legacyAssertions};}catch(error){return {id:c.id,message:c.message,pass:false,error:String(error?.message||error),legacyAssertions:c.legacyAssertions};}});
    const results=[...baseResults,...extras];const failed=results.filter(r=>!r.pass);
    return {mode:'current-contract-quality-ux-expanded',total:results.length,passed:results.length-failed.length,failed:failed.length,results,errors:failed.map(r=>r.error?`${r.message}: ${r.error}`:r.message),inheritedContractCount:baseResults.length,migratedGroupCount:extras.length,migratedLegacyAssertionEquivalents:extras.reduce((n,r)=>n+(r.legacyAssertions||0),0)};
  }

  function v148EvaluateBrowserUiContracts(){
    const base=Array.isArray(globalThis.V144_UI_CONTRACTS)?globalThis.V144_UI_CONTRACTS:[];const contracts=[...base,...V148_EXTRA_UI_CONTRACTS];
    const results=contracts.map(c=>{try{return {id:c.id,message:c.message,pass:!!c.test()};}catch(error){return {id:c.id,message:c.message,pass:false,error:String(error?.message||error)};}});const failed=results.filter(r=>!r.pass);
    const report={mode:'browser-dom',total:results.length,passed:results.length-failed.length,failed:failed.length,results};globalThis.FEQUEST_V148_UI_CONTRACT=report;return report;
  }

  function v148EvaluateBrowserUiLayer(){
    const total=(globalThis.V144_UI_CONTRACTS||[]).length+V148_EXTRA_UI_CONTRACTS.length;
    if(NODE_SMOKE()) return {mode:'deferred-to-browser-dom',total,passed:null,failed:null,results:[],deferredMessages:[...(globalThis.V144_UI_CONTRACTS||[]).map(c=>String(c.message)),...V148_EXTRA_UI_CONTRACTS.map(c=>String(c.message))]};
    return v148EvaluateBrowserUiContracts();
  }

  globalThis.V148_EXTRA_UI_CONTRACTS=V148_EXTRA_UI_CONTRACTS;
  globalThis.v148CueAudit=v148CueAudit;

  globalThis.v148EvaluateBrowserUiLayer=v148EvaluateBrowserUiLayer;
})();
