  function v153DiagnosticHelperAudit(){
    const groups={
      productionRuntime:[...V153_DIAGNOSTIC_HELPER_INVENTORY.productionRuntime],
      releaseOnly:[...V153_DIAGNOSTIC_HELPER_INVENTORY.releaseOnly],
      compatibilityOnly:[...V153_DIAGNOSTIC_HELPER_INVENTORY.compatibilityOnly]
    };
    const all=Object.values(groups).flat();
    const unique=[...new Set(all)];
    const missing=all.filter(name=>typeof globalThis[name]!=='function');
    return {
      total:all.length,
      unique:unique.length,
      duplicateCount:all.length-unique.length,
      productionRuntime:groups.productionRuntime.length,
      releaseOnly:groups.releaseOnly.length,

    };

  function v153ProductionCallBoundaryAudit(){
    const forbidden=[...V153_DIAGNOSTIC_HELPER_INVENTORY.releaseOnly,...V153_DIAGNOSTIC_HELPER_INVENTORY.compatibilityOnly];
    const runtimeNames=['runV153SelfCheck','v153EvaluateCurrentContract',...V153_DIAGNOSTIC_HELPER_INVENTORY.productionRuntime];
    const violations=[];
    runtimeNames.forEach(owner=>{
      const fn=globalThis[owner];
      if(typeof fn!=='function'){
        violations.push({owner,callee:'<missing>'});
        return;
      }
      const src=String(fn);
      forbidden.forEach(callee=>{
        if(src.includes(callee)) violations.push({owner,callee});
      });
    });
    return {runtimeNames,forbiddenCount:forbidden.length,violations,ok:violations.length===0};
  }

  function v153EvaluateCurrentContract(){
    const base=v148EvaluateCurrentContract();
    const baseResults=(base.results||[]).map(r=>r.id==='version-schema'
      ? {...r,message:'Current version/schema contract',pass:PROFILE_SCHEMA_VERSION===5&&APP_VERSION==='v153',error:undefined}

      catch(error){return {id:c.id,message:c.message,pass:false,error:String(error?.message||error),legacyAssertions:c.legacyAssertions};}
    });
    const extras=v153AdditionalCurrentChecks().map(c=>{
      try{return {id:c.id,message:c.message,pass:!!c.test(),legacyAssertions:c.legacyAssertions};}
      catch(error){return {id:c.id,message:c.message,pass:false,error:String(error?.message||error),legacyAssertions:c.legacyAssertions};}
    });
    const results=[...baseResults,...migrated,...extras],failed=results.filter(r=>!r.pass);
    return {
      mode:'current-contract-diagnostic-runtime-boundary',
      total:results.length,passed:results.length-failed.length,failed:failed.length,results,
      errors:failed.map(r=>r.error?`${r.message}: ${r.error}`:r.message),
      inheritedThroughV148:baseResults.length,
      rewrittenV149ToV152Checks:migrated.length,
      newV153Checks:extras.length
