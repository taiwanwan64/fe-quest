function subjectBFinalReviewTargetV243(detail,reason){
  const base=bFinalRemediationTarget(detail?.studyMode,detail?.sourceId,detail?.domain||'');
  if(detail?.kind!=='security'&&reason==='コード理解')return {mode:'compound',id:null};
  return base;
}

function applyBFinalReviewRouteV243(attempt){
  const wrong=(attempt?.details||[]).filter(d=>!d.ok);
  const buttons=[...document.querySelectorAll('[data-bfinalstudy]')];
  buttons.forEach((btn,i)=>{
    const detail=wrong[i];
    if(!detail)return;
    const key=bFinalMistakeKey(detail);
    const reason=profile.bFinalMistakeStats?.[key]?.lastReason||'';
    const target=subjectBFinalReviewTargetV243(detail,reason);
    if(target.mode!=='compound'||reason!=='コード理解'||detail.kind==='security')return;
    btn.onclick=()=>{
      document.getElementById('bFinalResult')?.classList.remove('show');
      setBMode('mock');
      startCompoundChallenge();
    };
    btn.dataset.bfinalRouteV243='compound';
  });
}
