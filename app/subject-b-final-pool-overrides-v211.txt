const __validateSubjectBSemanticsBeforeV211=validateSubjectBSemantics;
validateSubjectBSemantics=function(){
  const previous=__validateSubjectBSemanticsBeforeV211();
  const errors=(previous.errors||[]).filter(e=>e!=='exam algorithm pool must be 40'&&!String(e).startsWith('exam answer-position bias:'));
  const examPos=[0,0,0,0];
  B_EXAM_ALGO_ITEMS.forEach(q=>{if(Number.isInteger(q.a)&&q.a>=0&&q.a<4)examPos[q.a]++;});
  if(B_EXAM_ALGO_ITEMS.length!==43)errors.push(`exam algorithm pool must be 43: ${B_EXAM_ALGO_ITEMS.length}`);
  if(new Set(B_EXAM_ALGO_ITEMS.map(q=>q.id)).size!==B_EXAM_ALGO_ITEMS.length)errors.push('exam algorithm pool duplicate id');
  if(Math.max(...examPos)-Math.min(...examPos)>1)errors.push(`exam answer-position balance: ${examPos.join('/')}`);
  SUBJECT_B_FINAL_POOL_V211_ITEMS.forEach(q=>{
    if(B_EXAM_ALGO_CONTRACTS[q.id]!==String(q.options[q.a]))errors.push(`${q.id}: v211 exam semantic drift`);
  });
  const answerPositions={...(previous.answerPositions||{}),exam:examPos};
  return {...previous,ok:errors.length===0,errors,answerPositions};
};
