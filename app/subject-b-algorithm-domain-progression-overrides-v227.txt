function subjectBAlgorithmDomainEvidenceV227(){
  const rows={};
  for(const item of B_EXAM_ALGO_ITEMS){
    const d=makeFinalAlgoExam(item);
    const domain=d.domain||item.domain||'擬似言語';
    if(!rows[domain])rows[domain]={domain,misses:0,seen:0,distinctMissedItems:0};
    const row=rows[domain];
    const key=bFinalMistakeKey(d);
    const misses=Math.max(0,Number(profile.bFinalMistakeStats?.[key]?.misses)||0);
    const seen=Math.max(0,Number(profile.bFinalStats?.[`algo:${d.sourceId}`]?.seen)||0);
    row.misses+=misses;
    row.seen+=seen;
    if(misses>0)row.distinctMissedItems++;
  }
  const ranked=Object.values(rows).map(row=>{
    const evidenceSeen=Math.max(row.seen,row.misses);
    const errorRate=evidenceSeen?Math.min(100,Math.round(row.misses/evidenceSeen*100)):0;
    return {...row,evidenceSeen,errorRate};
  }).filter(row=>row.misses>0).sort((a,b)=>b.misses-a.misses||b.errorRate-a.errorRate||a.domain.localeCompare(b.domain,'ja'));
  const totalMistakes=ranked.reduce((sum,row)=>sum+row.misses,0);
  const top=ranked[0]||null,second=ranked[1]||null;
  const share=top&&totalMistakes?Math.round(top.misses/totalMistakes*100):0;
  const clearLead=!!top&&(!second||top.misses>second.misses||(top.misses===second.misses&&top.errorRate-second.errorRate>=SUBJECT_B_ALGORITHM_DOMAIN_PROGRESSION_V227_SPEC.equalMistakeCountRateLeadPercent));
  const qualifies=!!top
    &&totalMistakes>=SUBJECT_B_ALGORITHM_DOMAIN_PROGRESSION_V227_SPEC.minTotalMistakes
    &&top.misses>=SUBJECT_B_ALGORITHM_DOMAIN_PROGRESSION_V227_SPEC.minDomainMistakes
    &&top.distinctMissedItems>=SUBJECT_B_ALGORITHM_DOMAIN_PROGRESSION_V227_SPEC.minDistinctItems
    &&top.errorRate>=SUBJECT_B_ALGORITHM_DOMAIN_PROGRESSION_V227_SPEC.minDomainErrorRatePercent
    &&share>=SUBJECT_B_ALGORITHM_DOMAIN_PROGRESSION_V227_SPEC.minDomainMistakeSharePercent
    &&clearLead;
  return {totalMistakes,share,clearLead,qualifies,top,second,ranked};
}

function subjectBAlgorithmDomainTargetV227(domain){
  const item=B_EXAM_ALGO_ITEMS.find(x=>x.domain===domain);
  if(!item)return null;
  const d=makeFinalAlgoExam(item);
  const target=bFinalRemediationTarget(d.studyMode,d.sourceId,d.domain);
  if(target?.mode!=='trace'||!target.id||!B_EXERCISES.some(x=>x.id===target.id))return null;
  return {mode:'trace',id:target.id,domain:d.domain||domain};
}

  const first=profile.bFinalHistory?.[0];

  const firstTarget=subjectBFirstFinalTargetV222(first);
  if(firstTarget?.mode!=='miniMock')return null;
  const evidence=subjectBAlgorithmDomainEvidenceV227();

  const target=subjectBAlgorithmDomainTargetV227(evidence.top.domain);
  if(!target)return null;
  return {first,firstFinalKey:subjectBFirstFinalKeyV222(first),evidence,target};
}

    marker.completed=true;

    profile.subjectBAlgorithmDomainV227=subjectBAlgorithmDomainMarkerV227(context,true);

  }

}
